cmake_minimum_required(VERSION 3.24)
project(MirrorDeckNative LANGUAGES CXX)
add_library(MirrorDeck.NrBridge SHARED NrBridge.cpp)
set_target_properties(MirrorDeck.NrBridge PROPERTIES OUTPUT_NAME "nvngx.dll_MirrorDeckNR")
target_compile_features(MirrorDeck.NrBridge PRIVATE cxx_std_20)
set_property(TARGET MirrorDeck.NrBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(MirrorDeck.NrBridge PRIVATE vendor/dlss/include)
add_library(MirrorDeck.NativeDlss SHARED NativeDlss.cpp)
target_compile_features(MirrorDeck.NativeDlss PRIVATE cxx_std_20)
set_property(TARGET MirrorDeck.NativeDlss PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(MirrorDeck.NativeDlss PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(MirrorDeck.NativeDlss PRIVATE vendor/dlss/include vendor/nvof)
target_link_libraries(MirrorDeck.NativeDlss PRIVATE MirrorDeck.NrBridge "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dlss/lib/nvsdk_ngx_s.lib" d3d12 dxgi d3dcompiler dxguid shlwapi version)
target_compile_options(MirrorDeck.NativeDlss PRIVATE /W4 /EHsc /Zi)
target_link_options(MirrorDeck.NativeDlss PRIVATE /DEBUG)
